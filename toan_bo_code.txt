using System;
using System.Windows.Media;

namespace BleachVsNarutoMini.Engine
{
    public class GameLoop
    {
        private bool _isRunning;
        public event Action? Update;

        public void Start()
        {
            if (_isRunning) return;
            _isRunning = true;
            CompositionTarget.Rendering += OnRendering;
        }

        public void Stop()
        {
            if (!_isRunning) return;
            _isRunning = false;
            CompositionTarget.Rendering -= OnRendering;
        }

        private void OnRendering(object? sender, EventArgs e)
        {
            if (!_isRunning) return;
            Update?.Invoke();
        }
    }
}

    public static class InputManager
    {
        private static readonly HashSet<Key> PressedKeys = new HashSet<Key>();

        public static void KeyDown(Key key)
        {
            if (!PressedKeys.Contains(key))
                PressedKeys.Add(key);
        }

        public static void KeyUp(Key key)
        {
            if (PressedKeys.Contains(key))
                PressedKeys.Remove(key);
        }

        public static bool IsKeyDown(Key key)
        {
            return PressedKeys.Contains(key);
        }
    }
}

                Text = "100 / 100"
            };

            Children.Add(_bar);
            Children.Add(_text);
        }

        public void SetValue(double value, double max)
        {
            if (max <= 0) max = 1;
            value = Math.Max(0, Math.Min(value, max));

            _bar.Maximum = max;
            _bar.Value = value;

            // Xanh > 50%, vÃ ng > 25%, Ä‘á» cÃ²n láº¡i
            double pct = value / max;
            Color c = pct > 0.5 ? Color.FromRgb(0x4C, 0xAF, 0x50)
                    : pct > 0.25 ? Color.FromRgb(0xFF, 0xC1, 0x07)
                    : Color.FromRgb(0xF4, 0x43, 0x36);
            _bar.Foreground = new SolidColorBrush(c);

            _text.Text = (int)Math.Ceiling(value) + " / " + (int)max;
        }
    }

    public static class MapRenderer
    {
        public static void Draw(Canvas layer, MapData map)
        {
            layer.Children.Clear();
            layer.Background = map.Background;

            var gridBrush = new SolidColorBrush(Color.FromArgb(0x14, 255, 255, 255));

            // LÆ°á»›i má» trÃªn ná»n cho dá»… nhÃ¬n
            for (double x = 64; x < MapLibrary.Width; x += 64)
                layer.Children.Add(new Line
                {
                    X1 = x, X2 = x, Y1 = 0, Y2 = MapLibrary.Height,
                    Stroke = gridBrush, StrokeThickness = 1
                });
            for (double y = 64; y < MapLibrary.Height; y += 64)
                layer.Children.Add(new Line
                {
                    X1 = 0, X2 = MapLibrary.Width, Y1 = y, Y2 = y,
                    Stroke = gridBrush, StrokeThickness = 1
                });

            // Váº¡ch giá»¯a sÃ¢n
            layer.Children.Add(new Line
            {
                X1 = MapLibrary.Width / 2, X2 = MapLibrary.Width / 2,
                Y1 = 0, Y2 = MapLibrary.Height,
                Stroke = new SolidColorBrush(Color.FromArgb(0x30, 255, 255, 255)),
                StrokeThickness = 2,
                StrokeDashArray = new DoubleCollection { 6, 6 }
            });

            // Há»™p vÃ  tÆ°á»ng
            foreach (var r in map.Obstacles)
            {
                var box = new Rectangle
                {
                    Width = r.Width,
                    Height = r.Height,
                    Fill = map.ObstacleFill,
                    Stroke = map.ObstacleBorder,
                    StrokeThickness = 2,
                    RadiusX = 3,
                    RadiusY = 3
                };
                Canvas.SetLeft(box, r.X);
                Canvas.SetTop(box, r.Y);
                layer.Children.Add(box);
            }
        }
    }

        private static Style _buttonStyle;

        public static Style ButtonStyle
        {
            get
            {
                if (_buttonStyle == null)
                    _buttonStyle = (Style)XamlReader.Parse(ButtonStyleXaml);
                return _buttonStyle;
            }
        }

        public static Button MakeButton(string text, RoutedEventHandler onClick, Thickness margin)
        {
            var b = new Button
            {
                Content = text,
                Height = 45,
                Margin = margin,
                Style = ButtonStyle
            };
            b.Click += onClick;
            return b;
        }

        public static TextBlock MakeLabel(string text)
        {
            return new TextBlock
            {
                Text = text,
                Foreground = Brushes.White,
                FontSize = 16,
                Margin = new Thickness(0, 0, 0, 8)
            };
        }

        public static RadioButton MakeRadio(string text, string group, bool isChecked, Thickness margin)
        {
            return new RadioButton
            {
                Content = text,
                GroupName = group,
                IsChecked = isChecked,
                Foreground = Brushes.White,
                FontSize = 15,
                Margin = margin
            };
        }

        public static Brush DimBackground()
        {
            return new SolidColorBrush(Color.FromArgb(0xCC, 0, 0, 0));
        }
    }

using System;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Input;
using System.Windows.Media;
using System.Windows.Media.Imaging;
using System.Windows.Threading;

namespace Naruto
{
    public partial class MainWindow : Window
    {
        DispatcherTimer GameTimer = new DispatcherTimer();

        // --- BIẾN CỦA NARUTO ---
        int narutoRunFrame = 2;
        int narutoAttackFrame = 1;
        int narutoJumpFrame = 1;
        int narutoAttackTick = 0;
        bool narutoIsMoveLeft = false;
        bool narutoIsMoveRight = false;
        bool narutoIsAttacking = false;
        bool narutoIsJumping = false;

        // --- BIẾN CỦA SASUKE ---
        int sskRunFrame = 1;
        int sskAttackFrame = 1;
        int sskComboCount = 0;
        int sskJumpFrame = 1;
        int sskAttackTick = 0;
        bool sskIsMoveLeft = false;
        bool sskIsMoveRight = false;
        bool sskIsAttacking = false;
        bool sskIsJumping = false;

        public MainWindow()
        {
            InitializeComponent();
            GameTimer.Tick += GameTimer_Tick;
            GameTimer.Interval = TimeSpan.FromMilliseconds(50);
            GameTimer.Start();
        }

        private void GameTimer_Tick(object sender, EventArgs e)
        {
            // ================== LOGIC NARUTO ==================
            if (narutoIsAttacking)
            {
                narutoAttackTick++;
                if (narutoAttackTick >= 2)
                {
                    Player1.Source = new BitmapImage(new Uri("pack://application:,,,/Assets/danh" + narutoAttackFrame + ".png"));
                    narutoAttackFrame++;
                    narutoAttackTick = 0;

                    if (narutoAttackFrame > 3) // Đánh từ danh1 đến danh3
                    {
                        narutoIsAttacking = false;
                        narutoAttackFrame = 1;
                        Player1.Source = new BitmapImage(new Uri("pack://application:,,,/Assets/dung1.png"));
                    }
                }
            }
            else if (narutoIsJumping)
            {
                // Nhảy từ nhay1 đến nhay4 (2 hình đầu bay lên, 2 hình sau rớt xuống)
                if (narutoJumpFrame <= 2)
                    Canvas.SetTop(Player1, Canvas.GetTop(Player1) - 20);
                else
                    Canvas.SetTop(Player1, Canvas.GetTop(Player1) + 20);

                Player1.Source = new BitmapImage(new Uri("pack://application:,,,/Assets/nhay" + narutoJumpFrame + ".png"));
                narutoJumpFrame++;
                if (narutoJumpFrame > 4) // Đã sửa thành nhay4 theo đúng ý bạn
                {
                    narutoIsJumping = false;
                    narutoJumpFrame = 1;
                    Player1.Source = new BitmapImage(new Uri("pack://application:,,,/Assets/dung1.png"));
                }
            }
            else if (narutoIsMoveLeft)
            {
                Canvas.SetLeft(Player1, Canvas.GetLeft(Player1) - 10);
                Player1.RenderTransform = new ScaleTransform(-1, 1);
                Player1.RenderTransformOrigin = new Point(0.5, 0.5);

                if (narutoRunFrame < 2 || narutoRunFrame > 3) narutoRunFrame = 2;
                Player1.Source = new BitmapImage(new Uri("pack://application:,,,/Assets/dung" + narutoRunFrame + ".png"));
                narutoRunFrame = (narutoRunFrame == 2) ? 3 : 2;
            }
            else if (narutoIsMoveRight)
            {
                Canvas.SetLeft(Player1, Canvas.GetLeft(Player1) + 10);
                Player1.RenderTransform = new ScaleTransform(1, 1);
                Player1.RenderTransformOrigin = new Point(0.5, 0.5);

                if (narutoRunFrame < 2 || narutoRunFrame > 3) narutoRunFrame = 2;
                Player1.Source = new BitmapImage(new Uri("pack://application:,,,/Assets/dung" + narutoRunFrame + ".png"));
                narutoRunFrame = (narutoRunFrame == 2) ? 3 : 2;
            }

            // ================== LOGIC SASUKE ==================
            if (sskIsAttacking)
            {
                sskAttackTick++;
                if (sskAttackTick >= 2)
                {
                    Player2.Source = new BitmapImage(new Uri("pack://application:,,,/Assets/sskdanh" + sskAttackFrame + ".png"));
                    sskAttackFrame++;
                    sskAttackTick = 0;

                    if (sskComboCount == 1 && sskAttackFrame > 3)
                    {
                        sskIsAttacking = false;
                        sskAttackFrame = 1;
                        sskComboCount = 0;
                        Player2.Source = new BitmapImage(new Uri("pack://application:,,,/Assets/sskdung1.png"));
                    }
                    else if (sskComboCount >= 2 && sskAttackFrame > 7)
                    {
                        sskIsAttacking = false;
                        sskAttackFrame = 1;
                        sskComboCount = 0;
                        Player2.Source = new BitmapImage(new Uri("pack://application:,,,/Assets/sskdung1.png"));
                    }
                }
            }
            else if (sskIsJumping)
            {
                if (sskJumpFrame <= 3)
                    Canvas.SetTop(Player2, Canvas.GetTop(Player2) - 20);
                else
                    Canvas.SetTop(Player2, Canvas.GetTop(Player2) + 20);

                Player2.Source = new BitmapImage(new Uri("pack://application:,,,/Assets/ssknhay" + sskJumpFrame + ".png"));
                sskJumpFrame++;
                if (sskJumpFrame > 6)
                {
                    sskIsJumping = false;
                    sskJumpFrame = 1;
                    Player2.Source = new BitmapImage(new Uri("pack://application:,,,/Assets/sskdung1.png"));
                }
            }
            else if (sskIsMoveLeft)
            {
                Canvas.SetLeft(Player2, Canvas.GetLeft(Player2) - 10);
                Player2.RenderTransform = new ScaleTransform(-1, 1);
                Player2.RenderTransformOrigin = new Point(0.5, 0.5);
                Player2.Source = new BitmapImage(new Uri("pack://application:,,,/Assets/sskchay" + sskRunFrame + ".png"));
                sskRunFrame++;
                if (sskRunFrame > 6) sskRunFrame = 1;
            }
            else if (sskIsMoveRight)
            {
                Canvas.SetLeft(Player2, Canvas.GetLeft(Player2) + 10);
                Player2.RenderTransform = new ScaleTransform(1, 1);
                Player2.RenderTransformOrigin = new Point(0.5, 0.5);
                Player2.Source = new BitmapImage(new Uri("pack://application:,,,/Assets/sskchay" + sskRunFrame + ".png"));
                sskRunFrame++;
                if (sskRunFrame > 6) sskRunFrame = 1;
            }
        }

        private void Window_KeyDown(object sender, KeyEventArgs e)
        {
            if (e.Key == Key.A && !narutoIsAttacking && !narutoIsJumping) narutoIsMoveLeft = true;
            if (e.Key == Key.D && !narutoIsAttacking && !narutoIsJumping) narutoIsMoveRight = true;
            if (e.Key == Key.Space && !narutoIsAttacking && !narutoIsJumping) narutoIsJumping = true; // Phím Space nhảy
            if (e.Key == Key.J && !narutoIsJumping)
            {
                narutoIsAttacking = true;
                narutoAttackTick = 0;
            }

            if (e.Key == Key.Left && !sskIsAttacking && !sskIsJumping) sskIsMoveLeft = true;
            if (e.Key == Key.Right && !sskIsAttacking && !sskIsJumping) sskIsMoveRight = true;
            if (e.Key == Key.D5 && !sskIsAttacking && !sskIsJumping) sskIsJumping = true;
            if (e.Key == Key.D4 && !sskIsJumping)
            {
                sskIsAttacking = true;
                sskComboCount++;
                if (sskComboCount > 2) sskComboCount = 2;
            }
        }

        private void Window_KeyUp(object sender, KeyEventArgs e)
        {
            if (e.Key == Key.A)
            {
                narutoIsMoveLeft = false;
                narutoRunFrame = 2;
                if (!narutoIsAttacking && !narutoIsJumping)
                    Player1.Source = new BitmapImage(new Uri("pack://application:,,,/Assets/dung1.png"));
            }
            if (e.Key == Key.D)
            {
                narutoIsMoveRight = false;
                narutoRunFrame = 2;
                if (!narutoIsAttacking && !narutoIsJumping)
                    Player1.Source = new BitmapImage(new Uri("pack://application:,,,/Assets/dung1.png"));
            }

            if (e.Key == Key.Left)
            {
                sskIsMoveLeft = false;
                sskRunFrame = 1;
                if (!sskIsAttacking && !sskIsJumping)
                    Player2.Source = new BitmapImage(new Uri("pack://application:,,,/Assets/sskdung1.png"));
            }
            if (e.Key == Key.Right)
            {
                sskIsMoveRight = false;
                sskRunFrame = 1;
                if (!sskIsAttacking && !sskIsJumping)
                    Player2.Source = new BitmapImage(new Uri("pack://application:,,,/Assets/sskdung1.png"));
            }
        }
    }
}
